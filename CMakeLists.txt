cmake_minimum_required(VERSION 3.22)
project(VocalPilot VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(VOCALPILOT_DSP_ONLY "Build DSP tests without JUCE" OFF)
enable_testing()
add_executable(dsp_tests tests/DspTests.cpp)
target_include_directories(dsp_tests PRIVATE Source)
add_test(NAME dsp_tests COMMAND dsp_tests)
add_executable(tracking_tests tests/TrackingTests.cpp)
target_include_directories(tracking_tests PRIVATE Source)
add_test(NAME tracking_tests COMMAND tracking_tests)
add_executable(tracking_benchmark tests/TrackingBenchmark.cpp)
target_include_directories(tracking_benchmark PRIVATE Source)
add_executable(detector_comparison tests/DetectorComparison.cpp)
target_include_directories(detector_comparison PRIVATE Source/dsp)
add_executable(realtime_safety_tests tests/RealtimeSafetyTests.cpp)
target_include_directories(realtime_safety_tests PRIVATE Source)
add_test(NAME realtime_safety_tests COMMAND realtime_safety_tests)
add_executable(transformation_tests tests/TransformationTests.cpp)
target_include_directories(transformation_tests PRIVATE Source)
add_test(NAME transformation_tests COMMAND transformation_tests)
add_executable(transformation_benchmark tests/TransformationBenchmark.cpp)
target_include_directories(transformation_benchmark PRIVATE Source)
if(VOCALPILOT_DSP_ONLY)
  return()
endif()
set(JUCE_DIR "" CACHE PATH "Optional local JUCE checkout")
if(JUCE_DIR)
  add_subdirectory("${JUCE_DIR}" "${CMAKE_BINARY_DIR}/JUCE")
else()
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
    GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850 GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(JUCE)
endif()
juce_add_plugin(VocalPilot COMPANY_NAME "VocalPilot" BUNDLE_ID "audio.vocalpilot.prototype"
  PLUGIN_MANUFACTURER_CODE Vpjt PLUGIN_CODE Vp01 FORMATS VST3
  PRODUCT_NAME "VocalPilot" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE
  NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(VocalPilot PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(VocalPilot PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
  JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(VocalPilot PRIVATE juce::juce_audio_utils
  PUBLIC juce::juce_recommended_config_flags)
add_executable(processor_tests tests/ProcessorTests.cpp)
target_link_libraries(processor_tests PRIVATE VocalPilot)
target_include_directories(processor_tests PRIVATE Source "$<TARGET_PROPERTY:juce::juce_audio_utils,INTERFACE_INCLUDE_DIRECTORIES>")
add_test(NAME processor_tests COMMAND processor_tests)
add_executable(live_smoke tests/LiveSmoke.cpp)
target_link_libraries(live_smoke PRIVATE VocalPilot)
target_include_directories(live_smoke PRIVATE Source "$<TARGET_PROPERTY:juce::juce_audio_utils,INTERFACE_INCLUDE_DIRECTORIES>")
add_test(NAME live_smoke COMMAND live_smoke)
juce_add_console_app(vocalpilot_analyse PRODUCT_NAME "VocalPilot WAV Analysis")
target_sources(vocalpilot_analyse PRIVATE tools/AnalyseWav.cpp)
target_include_directories(vocalpilot_analyse PRIVATE Source)
target_compile_definitions(vocalpilot_analyse PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(vocalpilot_analyse PRIVATE juce::juce_audio_formats juce::juce_recommended_config_flags)
add_test(NAME wav_analysis COMMAND vocalpilot_analyse "${CMAKE_CURRENT_SOURCE_DIR}/examples/input-432.wav" "${CMAKE_CURRENT_BINARY_DIR}/analysis-smoke.csv" 0 major 100)
find_package(Python3 COMPONENTS Interpreter QUIET)
juce_add_console_app(vocalpilot_render PRODUCT_NAME "VocalPilot Engine Comparison")
target_sources(vocalpilot_render PRIVATE tools/RenderComparisons.cpp)
target_include_directories(vocalpilot_render PRIVATE Source)
target_compile_definitions(vocalpilot_render PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(vocalpilot_render PRIVATE juce::juce_audio_formats juce::juce_recommended_config_flags)
if(Python3_Interpreter_FOUND)
  execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import numpy" RESULT_VARIABLE VOCALPILOT_NUMPY OUTPUT_QUIET ERROR_QUIET)
  if(VOCALPILOT_NUMPY EQUAL 0)
    add_test(NAME transformation_quality_regression COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/transformation_quality.py" "$<TARGET_FILE:vocalpilot_render>" "${CMAKE_CURRENT_BINARY_DIR}/quality-regression" --quick --verify)
  endif()
  add_test(NAME render_validation COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/RenderHarnessTests.py" "$<TARGET_FILE:vocalpilot_render>")
  add_test(NAME wav_analysis_validation COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/HarnessTests.py" "$<TARGET_FILE:vocalpilot_analyse>")
endif()
option(VOCALPILOT_HOST_TEST "Build a VST3 binary smoke-test host" OFF)
if(VOCALPILOT_HOST_TEST)
  juce_add_console_app(vst3_smoke PRODUCT_NAME "VocalPilot VST3 Smoke Test")
  target_sources(vst3_smoke PRIVATE tests/Vst3Smoke.cpp)
  target_compile_definitions(vst3_smoke PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  target_link_libraries(vst3_smoke PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
  add_test(NAME vst3_smoke COMMAND vst3_smoke "$<TARGET_FILE:VocalPilot_VST3>")
endif()
